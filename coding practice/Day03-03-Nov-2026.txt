Anisble - Install Apache and Start Service
-----------------------------------------------------------------------------------------------------------------------

- name: Install and Start Apache
  hosts: all 
  become: yes
  task:
      - name: Install Apache
      	yum:
	   name: httpd
	   state: present
      
      - name: Start Apache
        service:
	   name: httpd
	   state: started
	   enable: yes

 Kubernetes — Create a ReplicaSet with 3 Replicas
-----------------------------------------------------------------------------------------------------------------------

apiVersion: apps/v1
kind: ReplicaSet
metadata:
    name: pathnex-replicate
spec: 
    replicas: 3 
    selector:
        matchLables:
            app: pathnex-app
    template:
        metadata:
	    lables:
	        app: pathnex-app
        spec:
	    containers:
	        - name: app
		  image: nginx
		  ports:
		      - containerPort: 80
